def detect_table_relationships(table_columns):

    prompt = f"""
You are a database expert.

Find relationships between tables using column names.

Return ONLY JSON like:
{{
  "relationships": [
    {{"table1":"table_name","column1":"column","table2":"table_name","column2":"column"}}
  ]
}}

Tables and columns:
{json.dumps(table_columns, indent=2)}
"""

    try:
        from model.llm_client import call_llm_chat
        messages = [{"role": "user", "content": prompt}]
        text = call_llm_chat(messages, json_mode=True, temperature=0.1, scenario="analysis")
        
        data = json.loads(text)
        return data.get("relationships", [])

    except Exception as e:
        print("Relationship detection error:", e)
        return []


def generate_session_graph(session_id, web_data, db_data, target_arango_db=None):

    net = Network(
        height="850px",
        width="100%",
        bgcolor="#222222",
        font_color="white",
        directed=False
    )

    SESSION_STYLE = {"color": "#00bfa5", "size": 35}
    TOPIC_STYLE   = {"color": "#ff4081", "size": 20}
    DB_STYLE      = {"color": "#2979ff", "size": 25}
    TABLE_STYLE   = {"color": "#ffc107", "size": 15}

    session_node = f"session_{session_id}"
    net.add_node(session_node, label=f"Session {session_id}", **SESSION_STYLE)

    # -------------------------
    # Web Topics
    # -------------------------
    for w in web_data:

        topic = w["topic"]
        net.add_node(topic, label=topic, **TOPIC_STYLE)
        net.add_edge(session_node, topic)

        for item in w["items"]:
            title = item["title"][:40]
            net.add_node(title, label=title)
            net.add_edge(topic, title)


    # -------------------------
    # DB SECTION
    # -------------------------
    table_columns = {} 
    for db in db_data:

        db_name = db["external_database"]
        display_name = db.get("display_name", db_name)

        net.add_node(db_name, label=display_name, **DB_STYLE)
        net.add_edge(session_node, db_name)

        for table in db["tables"]:

            tname = table["table_name"]
            columns = table.get("columns", [])
            table_columns[tname] = columns
            # Table node
            net.add_node(
                tname,
                label=tname,
                shape="box",
                **TABLE_STYLE
            )

            net.add_edge(db_name, tname)

            # Column nodes
            sample_rows = table.get("sample_rows", [])
            for col in columns:
                col_node = f"{tname}.{col}"

                net.add_node(
                    col_node,
                    label=col,
                    color="#9ccc65",
                    size=10
                )

                net.add_edge(tname, col_node)
                
                # Add values as leaf nodes to make the graph rich!
                added_vals = set()
                for row in sample_rows:
                    val = row.get(col)
                    if val is not None and str(val).strip():
                        val_str = str(val).strip()
                        # Shorten if too long
                        if len(val_str) > 25:
                            val_str = val_str[:22] + "..."
                            
                        if val_str not in added_vals:
                            added_vals.add(val_str)
                            val_node = f"{col_node}_{val_str}"
                            net.add_node(val_node, label=val_str, color="#bcaaa4", size=5, shape="text")
                            net.add_edge(col_node, val_node)


    # -------------------------
    # Detect Table Relationships using LLM
    # -------------------------

    relationships = detect_table_relationships(table_columns)
    if not isinstance(relationships, list):
        print(f"Warning: Unexpected relationships format: {type(relationships)}")
        relationships = []

    for r in relationships:

        try:

            col1 = f"{r['table1']}.{r['column1']}"
            col2 = f"{r['table2']}.{r['column2']}"

            net.add_edge(
                col1,
                col2,
                color="#ff5252",
                width=4,
                label="relation"
            )

        except Exception as e:
            print("Relationship edge error:", e)

    # -------------------------
    # Save Graph
    # -------------------------
    html_filename = f"graph_{uuid.uuid4().hex[:8]}.html"
    html_path = os.path.join(GRAPH_FOLDER, html_filename)

    net.save_graph(html_path)

    # --- Sync to ArangoDB ---
    try:
        _sync_to_arango(session_id, net.nodes, net.edges, target_arango_db)
    except Exception as e:
        print(f"[ArangoDB] Sync failed: {e}")

    graph_url = f"{BASE_URL}/graphs/{html_filename}"

    return graph_url

def _safe_key(val: str) -> str:
    """ArangoDB _key allows only [a-zA-Z0-9_:.@()-]+"""
    return re.sub(r'[^a-zA-Z0-9_:.@()-]', '_', str(val))

def _sync_to_arango(session_id: str, nodes: list, edges: list, target_arango_db: str = None):
    print(f"[ArangoDB] Connecting to {ARANGO_HOST} ...")
    client = ArangoClient(hosts=ARANGO_HOST)
    sys_db = client.db('_system', username=ARANGO_USER, password=ARANGO_PASS)

    db_to_use = target_arango_db if target_arango_db else ARANGO_DB

    # Ensure database exists
    if not sys_db.has_database(db_to_use):
        sys_db.create_database(db_to_use)
    
    db = client.db(db_to_use, username=ARANGO_USER, password=ARANGO_PASS)

    # Ensure collections exist
    nodes_col_name = "session_nodes"
    edges_col_name = "session_edges"

    if not db.has_collection(nodes_col_name):
        db.create_collection(nodes_col_name)
    if not db.has_collection(edges_col_name):
        db.create_collection(edges_col_name, edge=True)

    # --- Auto Create Graph Profile ---
    graph_name = "SessionGraph"
    if not db.has_graph(graph_name):
        db.create_graph(
            graph_name,
            edge_definitions=[
                {
                    "edge_collection": edges_col_name,
                    "from_vertex_collections": [nodes_col_name],
                    "to_vertex_collections": [nodes_col_name]
                }
            ]
        )
    # ---------------------------------

    print("[DEBUG] _sync_to_arango: REACHED BACKUP BLOCK", flush=True)
    from database.kgraph_backup_service import backup_kgraph_collections
    bkp_result = backup_kgraph_collections(db, nodes_col_name, edges_col_name)
    print(f"[DEBUG] _sync_to_arango: BACKUP DONE -- skipped={bkp_result.get('skipped')}", flush=True)

    nodes_col = db.collection(nodes_col_name)
    edges_col = db.collection(edges_col_name)

    # Sync Nodes
    print(f"[ArangoDB] Syncing {len(nodes)} nodes...")
    for n in nodes:
        key = _safe_key(n["id"])
        doc = {
            "_key": key,
            "session_id": session_id,
            "label": n.get("label", ""),
            "original_id": n["id"],
            "type": n.get("type") or (n["id"].split('_')[0] if '_' in n["id"] else "unknown")
        }
        try:
            if nodes_col.has(key):
                nodes_col.update(doc)
            else:
                nodes_col.insert(doc)
        except Exception as e:
            print(f"[ArangoDB] skip node {key}: {e}")

    # Sync Edges
    print(f"[ArangoDB] Syncing {len(edges)} edges...")
    for e in edges:
        from_key = _safe_key(e["from"])
        to_key   = _safe_key(e["to"])
        edge_key = _safe_key(f"{from_key}_to_{to_key}")
        
        doc = {
            "_key": edge_key,
            "_from": f"{nodes_col_name}/{from_key}",
            "_to": f"{nodes_col_name}/{to_key}",
            "session_id": session_id,
            "label": e.get("label", "relation"),
            "title": e.get("title", "")
        }
        try:
            if edges_col.has(edge_key):
                edges_col.update(doc)
            else:
                edges_col.insert(doc)
        except Exception as ex:
            print(f"[ArangoDB] skip edge {edge_key}: {ex}")

    print("[ArangoDB] Sync Complete!")

def _fetch_web_data(session_id: str, topics: list, conn) -> list:
    """Fetch saved_web_results filtered by session_id + topic list."""
    if not topics:
        return []
    cursor = None
    results = []
    try:
        cursor = conn.cursor(dictionary=True)
        placeholders = ",".join(["%s"] * len(topics))
        cursor.execute(f"""
            SELECT topic, title, url, brief, saved_at
            FROM saved_web_results
            WHERE session_id = %s
              AND topic IN ({placeholders})
            ORDER BY topic, saved_at DESC
        """, [session_id] + topics)
        rows = cursor.fetchall()

        # Group by topic
        grouped = {}
        for r in rows:
            t = r["topic"]
            if t not in grouped:
                grouped[t] = []
            grouped[t].append({
                "title":    r["title"],
                "url":      r["url"],
                "brief":    r.get("brief", ""),
                "saved_at": str(r["saved_at"]) if r["saved_at"] else None
            })

        for topic, items in grouped.items():
            results.append({
                "source_type":  "web",
                "topic":        topic,
                "result_count": len(items),
                "items":        items
            })
    except Exception as e:
        print(f"[Analysis] web fetch error: {e}")
    finally:
        if cursor: cursor.close()
    return results

def _dynamic_inspect_and_aggregate_database(ext_cur, tables: list) -> dict:
    """
    100% Generic & Dynamic Master/Transaction profiling and entity resolution.
    Works for ANY domain (Healthcare, Ports, Logistics, Finance, Sales, Inventory, Custom):
    1. Dynamically detects Master/Dimension tables (has candidate ID key + descriptive Name/Title/Description).
    2. Builds ID -> Real Entity Name mapping dictionaries.
    3. Dynamically detects Transaction/Fact tables (has additive measures, timestamps, and foreign keys).
    4. Automatically aggregates transaction data using real master entity names.
    5. Computes exact totals, averages, mins, maxes, categorical breakdowns, and time trends.
    """
    profile = {
        "master_mappings": {},      # { table_name: { str(id): str(name) } }
        "master_info": {},          # { table_name: { "id_col": str, "name_col": str } }
        "verified_aggregates": [],  # Human-readable exact calculation strings
        "table_stats": {}           # { table_name: { "numeric": {...}, "dates": {...}, "rows": int } }
    }

    numeric_types = ['int', 'decimal', 'float', 'double', 'numeric', 'bigint', 'smallint', 'tinyint']
    date_types = ['date', 'datetime', 'timestamp']

    # Known non-additive ID/Code patterns to avoid summing as meaningful metrics
    ID_PATTERN = re.compile(r'(_id|_code|_key|_num|_no|id|code|uuid|kunnr|matnr|phone|zip|postal|year|month|day)$', re.IGNORECASE)
    NAME_PATTERN = re.compile(r'(name|title|desc|description|label|cname|maktx|party|customer|vendor|vessel|doctor|patient|item|product|account)', re.IGNORECASE)

    table_meta = {}

    # Step A: Profile every table structure
    for t in tables:
        if t == 'workspace_files':
            continue
        try:
            ext_cur.execute(f"SHOW COLUMNS FROM `{t}`")
            cols_info = ext_cur.fetchall()
            
            ext_cur.execute(f"SELECT COUNT(*) as cnt FROM `{t}`")
            row_cnt = (ext_cur.fetchone() or {}).get("cnt", 0)

            numeric_cols = []
            measure_cols = []
            id_cols = []
            name_cols = []
            date_cols = []
            text_cols = []

            for c in cols_info:
                col_name = c["Field"]
                col_type = (c.get("Type") or "").lower()

                is_num = any(nt in col_type for nt in numeric_types)
                is_dt = any(dt in col_type for dt in date_types)

                if is_num:
                    numeric_cols.append(col_name)
                    if not ID_PATTERN.search(col_name):
                        measure_cols.append(col_name)
                    else:
                        id_cols.append(col_name)
                elif is_dt:
                    date_cols.append(col_name)
                else:
                    text_cols.append(col_name)
                    if ID_PATTERN.search(col_name):
                        id_cols.append(col_name)
                    if NAME_PATTERN.search(col_name):
                        name_cols.append(col_name)

            table_meta[t] = {
                "row_cnt": row_cnt,
                "cols_info": cols_info,
                "numeric_cols": numeric_cols,
                "measure_cols": measure_cols,
                "id_cols": id_cols,
                "name_cols": name_cols,
                "date_cols": date_cols,
                "text_cols": text_cols
            }
        except Exception as e:
            print(f"[DynamicProfiler] table {t} meta error: {e}")

    # Step B: Identify Master / Dimension Tables & Build Entity Dictionaries
    for t, meta in table_meta.items():
        t_lower = t.lower()
        is_named_master = any(k in t_lower for k in ['master', 'dim', 'lookup', 'type', 'cat'])
        has_id_and_name = len(meta["id_cols"]) > 0 and len(meta["name_cols"]) > 0
        is_small_entity = meta["row_cnt"] <= 5000 and (has_id_and_name or is_named_master)

        if is_named_master or has_id_and_name or is_small_entity:
            # Pick best candidate ID and Name column
            id_col = meta["id_cols"][0] if meta["id_cols"] else (meta["cols_info"][0]["Field"] if meta["cols_info"] else None)
            name_col = meta["name_cols"][0] if meta["name_cols"] else None

            if not name_col:
                # Fallback to first non-id text column
                candidates = [c for c in meta["text_cols"] if c != id_col]
                if candidates:
                    name_col = candidates[0]

            if id_col and name_col and id_col != name_col:
                try:
                    ext_cur.execute(f"SELECT `{id_col}`, `{name_col}` FROM `{t}` WHERE `{id_col}` IS NOT NULL LIMIT 300")
                    rows = ext_cur.fetchall()
                    mapping = {}
                    for r in rows:
                        k = str(r[id_col]).strip()
                        v = str(r[name_col]).strip()
                        if k and v:
                            mapping[k] = v

                    if mapping:
                        profile["master_mappings"][t] = mapping
                        profile["master_info"][t] = {"id_col": id_col, "name_col": name_col}
                        print(f"[DynamicProfiler] Detected Master Table '{t}': mapped {len(mapping)} entities via `{id_col}` -> `{name_col}`")
                except Exception as me:
                    print(f"[DynamicProfiler] mapping fetch error on {t}: {me}")

    # Step C: Compute Verified Aggregations for Fact / Transaction Tables
    for t, meta in table_meta.items():
        row_cnt = meta["row_cnt"]
        if row_cnt == 0:
            continue

        table_verified = [f"Table `{t}` (Total Verified Rows: {row_cnt}):"]

        # 1. Global Measure Summaries (SUM, AVG, MIN, MAX)
        for m_col in meta["measure_cols"][:4]:
            try:
                ext_cur.execute(f"SELECT SUM(`{m_col}`) as sum_val, AVG(`{m_col}`) as avg_val, MIN(`{m_col}`) as min_val, MAX(`{m_col}`) as max_val FROM `{t}`")
                res = ext_cur.fetchone()
                if res and res.get("sum_val") is not None:
                    sum_v = round(float(res["sum_val"]), 2)
                    avg_v = round(float(res["avg_val"]), 2) if res["avg_val"] is not None else 0
                    min_v = round(float(res["min_val"]), 2) if res["min_val"] is not None else 0
                    max_v = round(float(res["max_val"]), 2) if res["max_val"] is not None else 0
                    table_verified.append(f"  • Metric `{m_col}`: Total SUM = {sum_v:,.2f}, Average = {avg_v:,.2f}, Range = [{min_v:,.2f} to {max_v:,.2f}]")
            except Exception as se:
                print(f"[DynamicProfiler] sum error `{t}`.`{m_col}`: {se}")

        # 2. Foreign Key Join to Master Tables -> Group By REAL Entity Name
        primary_measure = meta["measure_cols"][0] if meta["measure_cols"] else None

        for col in (meta["id_cols"] + meta["text_cols"]):
            col_lower = col.lower()
            matched_master = None

            for m_tbl, m_info in profile["master_info"].items():
                m_tbl_clean = re.sub(r'(_master|master|dim_|_dim|tbl_)', '', m_tbl.lower())
                m_id_col = m_info["id_col"].lower()

                if col_lower == m_id_col or col_lower == f"{m_tbl_clean}_id" or col_lower == f"{m_tbl_clean}_code" or (m_tbl_clean in col_lower and ("id" in col_lower or "code" in col_lower or "party" in col_lower or "cust" in col_lower)):
                    matched_master = m_tbl
                    break

            if matched_master and matched_master in profile["master_mappings"]:
                m_map = profile["master_mappings"][matched_master]
                try:
                    if primary_measure:
                        ext_cur.execute(f"""
                            SELECT `{col}` as fk_val, COUNT(*) as cnt, SUM(`{primary_measure}`) as total_m
                            FROM `{t}`
                            WHERE `{col}` IS NOT NULL
                            GROUP BY `{col}`
                            ORDER BY total_m DESC
                            LIMIT 10
                        """)
                    else:
                        ext_cur.execute(f"""
                            SELECT `{col}` as fk_val, COUNT(*) as cnt
                            FROM `{t}`
                            WHERE `{col}` IS NOT NULL
                            GROUP BY `{col}`
                            ORDER BY cnt DESC
                            LIMIT 10
                        """)
                    grp_rows = ext_cur.fetchall()
                    if grp_rows:
                        table_verified.append(f"  • Breakdown by Verified Master '{matched_master}' (via `{col}`):")
                        for gr in grp_rows:
                            fk_str = str(gr.get("fk_val")).strip()
                            resolved_name = m_map.get(fk_str, f"Entity_{fk_str}")
                            if primary_measure:
                                tot_val = round(float(gr.get("total_m") or 0), 2)
                                table_verified.append(f"      - '{resolved_name}': {tot_val:,.2f} (Transactions/Count: {gr['cnt']})")
                            else:
                                table_verified.append(f"      - '{resolved_name}': Count = {gr['cnt']}")
                except Exception as ge:
                    print(f"[DynamicProfiler] group join error on `{t}`.`{col}`: {ge}")

        # 3. Categorical Breakdown (for text columns with moderate cardinality)
        for cat_col in meta["text_cols"][:3]:
            if ID_PATTERN.search(cat_col):
                continue
            try:
                ext_cur.execute(f"SELECT COUNT(DISTINCT `{cat_col}`) as u_cnt FROM `{t}`")
                u_res = ext_cur.fetchone() or {}
                u_cnt = u_res.get("u_cnt", 0)

                if 1 < u_cnt <= 20:
                    if primary_measure:
                        ext_cur.execute(f"""
                            SELECT `{cat_col}` as val, COUNT(*) as cnt, SUM(`{primary_measure}`) as total_m
                            FROM `{t}`
                            WHERE `{cat_col}` IS NOT NULL AND TRIM(`{cat_col}`) != ''
                            GROUP BY `{cat_col}`
                            ORDER BY total_m DESC
                            LIMIT 8
                        """)
                    else:
                        ext_cur.execute(f"""
                            SELECT `{cat_col}` as val, COUNT(*) as cnt
                            FROM `{t}`
                            WHERE `{cat_col}` IS NOT NULL AND TRIM(`{cat_col}`) != ''
                            GROUP BY `{cat_col}`
                            ORDER BY cnt DESC
                            LIMIT 8
                        """)
                    cat_rows = ext_cur.fetchall()
                    if cat_rows:
                        table_verified.append(f"  • Category Breakdown by `{cat_col}`:")
                        for cr in cat_rows:
                            c_val = str(cr.get("val")).strip()
                            if primary_measure:
                                c_tot = round(float(cr.get("total_m") or 0), 2)
                                table_verified.append(f"      - '{c_val}': {c_tot:,.2f} (Count: {cr['cnt']})")
                            else:
                                table_verified.append(f"      - '{c_val}': Count = {cr['cnt']}")
            except Exception as ce:
                print(f"[DynamicProfiler] cat breakdown error on `{t}`.`{cat_col}`: {ce}")

        # 4. Time Trend (Date columns)
        if meta["date_cols"]:
            d_col = meta["date_cols"][0]
            try:
                ext_cur.execute(f"SELECT MIN(`{d_col}`) as min_d, MAX(`{d_col}`) as max_d FROM `{t}`")
                d_range = ext_cur.fetchone() or {}
                if d_range.get("min_d"):
                    table_verified.append(f"  • Date Timeline `{d_col}`: From {d_range['min_d']} to {d_range['max_d']}")

                if primary_measure:
                    ext_cur.execute(f"""
                        SELECT DATE_FORMAT(`{d_col}`, '%Y-%m') as period, SUM(`{primary_measure}`) as tot_val, COUNT(*) as cnt
                        FROM `{t}`
                        WHERE `{d_col}` IS NOT NULL
                        GROUP BY period
                        ORDER BY period ASC
                        LIMIT 12
                    """)
                    t_rows = ext_cur.fetchall()
                    if t_rows:
                        trend_items = [f"{tr['period']}: {round(float(tr.get('tot_val') or 0), 2):,.2f}" for tr in t_rows if tr.get('period')]
                        table_verified.append(f"  • Monthly Trend ({primary_measure}): " + ", ".join(trend_items))
            except Exception as de:
                print(f"[DynamicProfiler] date trend error on `{t}`: {de}")

        if len(table_verified) > 1:
            profile["verified_aggregates"].append("\n".join(table_verified))

    return profile


def _fetch_db_data(session_id: str, databaseses: list, conn) -> list:
    """
    Generic dynamic database fetcher:
    - Reads tables and sample data.
    - Dynamically detects Master and Transaction tables.
    - Performs verified SQL aggregations and entity name resolutions.
    """
    if not databaseses:
        return []

    cursor = None
    results = []

    try:
        cursor = conn.cursor(dictionary=True)
        placeholders = ",".join(["%s"] * len(databaseses))

        cursor.execute(f"""
            SELECT DISTINCT external_database, new_user_db
            FROM external_db_sync_log
            WHERE session_id = %s
              AND external_database IN ({placeholders})
              AND new_user_db IS NOT NULL
              AND new_user_db != ''
        """, [session_id] + databaseses)

        rows = cursor.fetchall()
        db_map = {r["external_database"]: r["new_user_db"] for r in rows}
        print(f"[Analysis] DB MAP -> {db_map}")

    except Exception as e:
        print(f"[Analysis] db_map error: {e}")
        return []
    finally:
        if cursor:
            cursor.close()

    unique_new_dbs = list(set(db_map.values()))

    for new_db in unique_new_dbs:
        mapped_ext_dbs = [ext for ext, ndb in db_map.items() if ndb == new_db]
        display_name_str = ", ".join(mapped_ext_dbs)

        db_result = {
            "source_type": "database",
            "external_database": display_name_str,
            "new_user_db": new_db,
            "tables": [],
            "dynamic_profile": {}
        }

        ext_conn = None
        ext_cur = None

        try:
            print(f"[Analysis] Connecting DB -> {new_db}")
            ext_conn = mysql.connector.connect(
                host=MYSQL_CONFIG["host"],
                port=MYSQL_CONFIG["port"],
                user=MYSQL_CONFIG["user"],
                password=MYSQL_CONFIG["password"],
                database=new_db,
                connection_timeout=10
            )

            ext_cur = ext_conn.cursor(dictionary=True)

            # Fetch table list
            ext_cur.execute("SHOW TABLES")
            tables_raw = ext_cur.fetchall()
            tables = [list(r.values())[0] for r in tables_raw]

            print(f"[Analysis] Tables in {new_db} -> {tables}")

            # Run 100% generic master & transaction dynamic profiling
            dyn_profile = _dynamic_inspect_and_aggregate_database(ext_cur, tables)
            db_result["dynamic_profile"] = dyn_profile

            for t in tables:
                print(f"[Analysis] Reading table -> {t}")
                try:
                    if t == 'workspace_files':
                        ext_cur.execute(f"SELECT * FROM `{t}` ORDER BY id DESC LIMIT %s", (MAX_ROWS,))
                    else:
                        ext_cur.execute(f"SELECT * FROM `{t}` LIMIT %s", (MAX_ROWS,))
                    rows = ext_cur.fetchall()

                    row_count = len(rows)
                    cols = list(rows[0].keys()) if row_count > 0 else []

                    # Column statistics
                    col_stats = {}
                    if row_count > 0:
                        for col in cols:
                            vals = [str(r.get(col)).strip() for r in rows if r.get(col) is not None and str(r.get(col)).strip()]
                            distinct_vals = list(dict.fromkeys(vals))
                            col_stats[col] = {
                                "total_values": len(vals),
                                "distinct_values": len(distinct_vals),
                                "sample": distinct_vals[:8]
                            }

                    if t == 'workspace_files':
                        for r in rows:
                            try:
                                import json
                                if isinstance(r.get('file_data'), str):
                                    file_data = json.loads(r['file_data'])
                                else:
                                    file_data = r.get('file_data') or {}
                                    
                                file_name = r.get('file_name', 'Document')
                                db_result["display_name"] = file_name
                                    
                                structured = file_data.get('structured_content')
                                if structured:
                                    paragraphs = structured.get('paragraphs', [])
                                    if paragraphs:
                                        db_result["tables"].append({
                                            "table_name": "Document_Paragraphs",
                                            "row_count": len(paragraphs),
                                            "columns": ["title", "text"],
                                            "column_stats": {},
                                            "sample_rows": paragraphs[:5]
                                        })
                                    for idx, vtable in enumerate(structured.get('tables', [])):
                                        vname = vtable.get('table_name', f"PDF_Table_{idx+1}")
                                        vcols = vtable.get('headers', [])
                                        sample_vrows = []
                                        for vrow in vtable.get('rows', [])[:5]:
                                            row_dict = {vcols[i]: vrow[i] for i in range(min(len(vcols), len(vrow)))}
                                            sample_vrows.append(row_dict)
                                            
                                        db_result["tables"].append({
                                            "table_name": vname,
                                            "row_count": len(vtable.get('rows', [])),
                                            "columns": vcols,
                                            "column_stats": {},
                                            "sample_rows": sample_vrows
                                        })
                            except Exception as e:
                                print(f"Error parsing workspace_files json: {e}")
                    else:
                        db_result["tables"].append({
                            "table_name": t,
                            "row_count": row_count,
                            "columns": cols,
                            "column_stats": col_stats,
                            "sample_rows": rows[:5] if rows else []
                        })

                except Exception as table_error:
                    print(f"[Analysis] table {t} error -> {table_error}")

        except Exception as db_error:
            print(f"[Analysis] connect {new_db} error -> {db_error}")

        finally:
            if ext_cur:
                try: ext_cur.close()
                except: pass
            if ext_conn:
                try: ext_conn.close()
                except: pass

        results.append(db_result)

    return results

# CONTEXT BUILDER
# ══════════════════════════════════════════════════════

def _build_context(web_data: list, db_data: list) -> str:
    parts = []

    # 1. Master Entity Mappings (Top Priority: ensures LLM resolves IDs to Real Entity Names)
    master_mappings_text = []
    for d in db_data:
        dyn = d.get("dynamic_profile", {})
        mappings = dyn.get("master_mappings", {})
        for m_tbl, m_dict in mappings.items():
            sample_entries = [f"ID '{k}' -> '{v}'" for k, v in list(m_dict.items())[:25]]
            master_mappings_text.append(f"Master Table `{m_tbl}` (Total Entities: {len(m_dict)}):\n  " + "\n  ".join(sample_entries))

    if master_mappings_text:
        parts.append(
            "=======================================================================\n"
            "=== 1. VERIFIED MASTER ENTITY MAPPINGS (MANDATORY TO USE IN CHART LABELS) ===\n"
            "=======================================================================\n"
            "CRITICAL: If a foreign key refers to any of these entities, ALWAYS use the\n"
            "descriptive Name from this mapping. NEVER output 'Part 1', 'Party 1', or 'Category 1'!\n\n"
            + "\n\n".join(master_mappings_text)
        )

    # 2. Verified Aggregations & Metrics (Exact numbers for KPIs and Charts)
    verified_aggs = []
    for d in db_data:
        dyn = d.get("dynamic_profile", {})
        aggs = dyn.get("verified_aggregates", [])
        if aggs:
            verified_aggs.extend(aggs)

    if verified_aggs:
        parts.append(
            "=======================================================================\n"
            "=== 2. VERIFIED REAL AGGREGATIONS & METRICS (USE EXACT FIGURES FOR KPIS & CHARTS) ===\n"
            "=======================================================================\n"
            "Use these verified database aggregations as the single source of truth for all\n"
            "KPI values, sums, totals, rankings, category breakdowns, and chart labels:\n\n"
            + "\n\n".join(verified_aggs)
        )

    # 3. Web Data (if any)
    for w in web_data:
        lines = [f"=== WEB TOPIC: {w['topic']} ({w['result_count']} results) ==="]
        for item in w["items"]:
            lines.append(f"  Title : {item['title']}")
            lines.append(f"  URL   : {item['url']}")
            if item.get("brief"):
                lines.append(f"  Brief : {item['brief'][:300]}")
        parts.append("\n".join(lines))

    # 4. Database Schema and Sample Rows
    for d in db_data:
        lines = [f"=== DATABASE SCHEMA: {d['external_database']} (stored as: {d['new_user_db']}) ==="]
        for tbl in d["tables"]:
            lines.append(f"\n  Table: {tbl['table_name']} ({tbl['row_count']} rows)")
            lines.append(f"  Columns: {', '.join(tbl['columns'])}")
            for col, stats in tbl.get("column_stats", {}).items():
                lines.append(
                    f"    {col}: {stats['distinct_values']} distinct values — sample: {', '.join(str(v) for v in stats['sample'][:6])}"
                )
            lines.append(f"  Sample rows (up to 5):")
            for i, row in enumerate(tbl.get("sample_rows", []), 1):
                r_str = " | ".join(f"{k}:{v}" for k, v in row.items() if v is not None and str(v).strip())
                lines.append(f"    Row{i}: {r_str}")
        parts.append("\n".join(lines))

    ctx = "\n\n".join(parts)
    if len(ctx) > MAX_CTX_CHARS:
        ctx = ctx[:MAX_CTX_CHARS] + "\n\n[... truncated ...]"
    return ctx
